const CORS = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Headers": "content-type", "Access-Control-Allow-Methods": "POST, OPTIONS" }; function json(obj, status) { return new Response(JSON.stringify(obj), { status: status, headers: Object.assign({ "content-type": "application/json" }, CORS) }); } export default { async fetch(request, env) { if (request.method === "OPTIONS") { return new Response(null, { headers: CORS }); } if (request.method !== "POST") { return json({ error: { message: "Metodo non consentito" } }, 405); } let body; try { body = await request.json(); } catch (e) { return json({ error: { message: "Richiesta non leggibile" } }, 400); } if (!env.APP_PASSWORD || body.password !== env.APP_PASSWORD) { return json({ error: { message: "Password non valida" } }, 401); } if (!env.ANTHROPIC_API_KEY) { return json({ error: { message: "Chiave API non configurata sul Worker" } }, 500); } const headers = { "content-type": "application/json", "x-api-key": env.ANTHROPIC_API_KEY, "anthropic-version": "2023-06-01" }; if (body.action === "models") { const r = await fetch("https://api.anthropic.com/v1/models?limit=100", { headers: headers }); return new Response(await r.text(), { status: r.status, headers: Object.assign({ "content-type": "application/json" }, CORS) }); } const model = body.model || env.MODEL; if (!model) { return json({ error: { message: "Nessun modello indicato" } }, 400); } const r = await fetch("https://api.anthropic.com/v1/messages", { method: "POST", headers: headers, body: JSON.stringify({ model: model, max_tokens: Math.min(Number(body.max_tokens) || Number(env.MAX_TOKENS) || 8000, 32000), system: body.system, messages: body.messages }) }); return new Response(await r.text(), { status: r.status, headers: Object.assign({ "content-type": "application/json" }, CORS) }); } };
